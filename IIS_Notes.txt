	USE [TaraSir_ImageBase];
	GO
	-- 1. Ensure the Windows Login exists on the server (if not already added)
	IF NOT EXISTS (SELECT * FROM sys.server_principals WHERE name = N'IIS APPPOOL\ImageBasedOMR')
	BEGIN
    	CREATE LOGIN [IIS APPPOOL\ImageBasedOMR] FROM WINDOWS;
	END
	GO
	-- 2. Ensure the user exists in the database
	IF NOT EXISTS (SELECT * FROM sys.database_principals WHERE name = N'IIS APPPOOL\ImageBasedOMR')
	BEGIN
    	CREATE USER [IIS APPPOOL\ImageBasedOMR] FOR LOGIN [IIS APPPOOL\ImageBasedOMR];
	END
	GO
	-- 3. Grant table creation permissions
	GRANT CREATE TABLE TO [IIS APPPOOL\ImageBasedOMR];
	GO
